% Deja estirar un poco las líneas antes que desbordar el margen: hay

% biblatex-ieee antepone a la serie de las actas (LNCS) la abreviatura
% «jourser», que en español es «ép.» (época). En IEEE es «ser.». Las fuentes
% web siguen la guía IEEE de la UMA: «Accedido: fecha. [En línea]. Disponible
% en: URL». Esta orden admite un solo valor por cadena.
\DefineBibliographyStrings{spanish}{
  jourser = {ser\adddot},
  url     = {[En línea]\adddot\addspace Disponible en},
  urlseen = {Accedido\addcolon},
}
